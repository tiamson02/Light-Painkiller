cmake_minimum_required(VERSION 3.21)
project(Light VERSION 0.1.7 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(light_core STATIC src/light_registry.cpp)
target_include_directories(light_core PUBLIC include)
include(CTest)
if(BUILD_TESTING)
  add_executable(light_tests tests/test_registry.cpp)
  target_link_libraries(light_tests PRIVATE light_core)
  add_test(NAME light_registry COMMAND light_tests)
endif()

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Light injects into a 32-bit engine: configure Visual Studio Win32 (-A Win32).")
  endif()
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG c3fcafd GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(minhook)
  set(PATCH_SOURCES src/patch.cpp src/lighting.cpp src/platform.cpp src/proxy.cpp)
  foreach(target IN ITEMS dinput8 Light)
    add_library(${target} SHARED ${PATCH_SOURCES})
    target_include_directories(${target} PRIVATE include src)
    target_link_libraries(${target} PRIVATE light_core minhook)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /EHsc)
    endif()
    set_target_properties(${target} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/release")
  endforeach()
  target_compile_definitions(dinput8 PRIVATE LIGHT_PROXY=1)
  target_sources(dinput8 PRIVATE src/dinput8.def)
  target_compile_definitions(Light PRIVATE LIGHT_PROXY=0)
  target_sources(Light PRIVATE src/Light.def)
endif()
